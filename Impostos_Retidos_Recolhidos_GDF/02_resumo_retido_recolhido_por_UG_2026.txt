-- =====================================================================
-- RESUMO: VALOR RETIDO x VALOR RECOLHIDO POR UG E TIPO DE IMPOSTO
-- Exercicio 2026 (schema MIL2026)
--
-- Regra de classificacao:
--   Contas "A PAGAR/RETIDO" (827110203, 631810000, 632110300): valor
--     retido = soma dos CREDITOS nessas contas.
--   Contas "PAGAS/RECOLHIDO" (827110204, 631820000, 632210300): valor
--     recolhido = soma dos DEBITOS nessas contas.
--
-- INCLUI EVENTOS DE ESTORNO (ver observacao no script 01).
-- TRAVA DE ANO nas contas 827110203/827110204 (ver observacao no script 01).
-- =====================================================================

WITH EVENTOS_TEXTO_2026 AS (
    SELECT DISTINCT er.COCONTACONTABIL, er.COEVENTO, e.NOEVENTO, e.TXDESCRICAO
    FROM MIL2026.EVENTOROTEIRO er
    JOIN MIL2026.EVENTO e ON e.COEVENTO = er.COEVENTO
    WHERE er.COCONTACONTABIL IN ('827110203','827110204','631810000','631820000','632110300','632210300')
      AND (
            UPPER(e.NOEVENTO) LIKE '%ISS%'  OR UPPER(e.TXDESCRICAO) LIKE '%ISS%'
         OR UPPER(e.NOEVENTO) LIKE '%ICMS%' OR UPPER(e.TXDESCRICAO) LIKE '%ICMS%'
         OR UPPER(e.NOEVENTO) LIKE '%IPTU%' OR UPPER(e.TXDESCRICAO) LIKE '%IPTU%'
         OR UPPER(e.NOEVENTO) LIKE '%IOF%'  OR UPPER(e.TXDESCRICAO) LIKE '%IOF%'
         OR UPPER(e.NOEVENTO) LIKE '%ITBI%' OR UPPER(e.TXDESCRICAO) LIKE '%ITBI%'
         OR UPPER(e.NOEVENTO) LIKE '%ITCD%' OR UPPER(e.TXDESCRICAO) LIKE '%ITCD%'
         OR UPPER(e.NOEVENTO) LIKE '%IRRF%' OR UPPER(e.TXDESCRICAO) LIKE '%IRRF%'
         OR UPPER(e.NOEVENTO) LIKE '%IRPF%' OR UPPER(e.TXDESCRICAO) LIKE '%IRPF%'
         OR UPPER(e.NOEVENTO) LIKE '%IRPJ%' OR UPPER(e.TXDESCRICAO) LIKE '%IRPJ%'
         OR UPPER(e.NOEVENTO) LIKE '%IMPOSTO DE RENDA%' OR UPPER(e.TXDESCRICAO) LIKE '%IMPOSTO DE RENDA%'
         OR (UPPER(e.NOEVENTO) LIKE '%IMPOSTO%' AND UPPER(e.NOEVENTO) NOT LIKE '%INSS%')
      )
      AND UPPER(e.NOEVENTO) NOT LIKE '%INSS%'
      AND UPPER(e.TXDESCRICAO) NOT LIKE '%INSS%'
),
EVENTOS_PARES_2026 AS (
    -- Inclui o evento PAR (principal <-> estorno) de cada evento de imposto ja identificado,
    -- mesmo que a descricao do par nao contenha as mesmas palavras-chave.
    -- Convencao SIGGO: 3o digito do codigo do evento: 0<->5 e 1<->6 sao pares principal/estorno.
    SELECT DISTINCT er2.COCONTACONTABIL, er2.COEVENTO, e2.NOEVENTO, e2.TXDESCRICAO
    FROM EVENTOS_TEXTO_2026 et
    JOIN MIL2026.EVENTOROTEIRO er2
      ON er2.COCONTACONTABIL = et.COCONTACONTABIL
     AND er2.COEVENTO = CASE SUBSTR(TO_CHAR(et.COEVENTO), 3, 1)
                              WHEN '0' THEN TO_NUMBER(SUBSTR(TO_CHAR(et.COEVENTO),1,2) || '5' || SUBSTR(TO_CHAR(et.COEVENTO),4))
                              WHEN '5' THEN TO_NUMBER(SUBSTR(TO_CHAR(et.COEVENTO),1,2) || '0' || SUBSTR(TO_CHAR(et.COEVENTO),4))
                              WHEN '1' THEN TO_NUMBER(SUBSTR(TO_CHAR(et.COEVENTO),1,2) || '6' || SUBSTR(TO_CHAR(et.COEVENTO),4))
                              WHEN '6' THEN TO_NUMBER(SUBSTR(TO_CHAR(et.COEVENTO),1,2) || '1' || SUBSTR(TO_CHAR(et.COEVENTO),4))
                            END
    JOIN MIL2026.EVENTO e2 ON e2.COEVENTO = er2.COEVENTO
),
EVENTOS_IMPOSTO_2026 AS (
    SELECT COCONTACONTABIL, COEVENTO, NOEVENTO, TXDESCRICAO FROM EVENTOS_TEXTO_2026
    UNION
    SELECT COCONTACONTABIL, COEVENTO, NOEVENTO, TXDESCRICAO FROM EVENTOS_PARES_2026
),
BASE AS (
    SELECT
        l.COUG,
        ug.NOUG,
        l.COCONTACONTABIL,
        ei.NOEVENTO,
        ei.TXDESCRICAO,
        l.INDEBITOCREDITO,
        l.VALANCAMENTO,
        CASE
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%ISS%'  THEN 'ISS'
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%ICMS%' THEN 'ICMS'
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%IPTU%' THEN 'IPTU'
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%IOF%'  THEN 'IOF'
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%ITBI%' THEN 'ITBI'
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%ITCD%' THEN 'ITCD'
            WHEN UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%IRRF%'
              OR UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%IRPF%'
              OR UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%IRPJ%'
              OR UPPER(ei.NOEVENTO || ' ' || ei.TXDESCRICAO) LIKE '%IMPOSTO DE RENDA%' THEN 'IR (IRPF/IRPJ)'
            ELSE 'OUTROS IMPOSTOS'
        END AS TIPO_IMPOSTO
    FROM MIL2026.LANCAMENTOCONTABIL l
    JOIN EVENTOS_IMPOSTO_2026 ei ON ei.COEVENTO = l.COEVENTO AND ei.COCONTACONTABIL = l.COCONTACONTABIL
    LEFT JOIN MIL2026.UNIDADEGESTORA ug ON ug.COUG = l.COUG
    WHERE l.COCONTACONTABIL NOT IN (827110203, 827110204)
       OR SUBSTR(TRIM(l.COCONTACORRENTE), 1, 4) = '2026'
)
SELECT
    COUG,
    NOUG,
    TIPO_IMPOSTO,
    SUM(CASE WHEN COCONTACONTABIL IN ('631810000','632110300','827110203') AND INDEBITOCREDITO = 'C'
             THEN VALANCAMENTO ELSE 0 END) AS VALOR_RETIDO,
    SUM(CASE WHEN COCONTACONTABIL IN ('631820000','632210300','827110204') AND INDEBITOCREDITO = 'D'
             THEN VALANCAMENTO ELSE 0 END) AS VALOR_RECOLHIDO
FROM BASE
GROUP BY COUG, NOUG, TIPO_IMPOSTO
ORDER BY COUG, TIPO_IMPOSTO;
